function openDrawer(row) { $("drawerTitle").textContent = row.symbol || "Stock Details"; $("drawerSubtitle").textContent = row.companyName || ""; const body = []; // ========================================================= // 1. TRIGGERS // ========================================================= body.push( detailSection( "Triggers", [ detailItem( "Fundamental Trigger", textOrDash(row.fundamentalTrigger) ), detailItem( "Key Trigger", textOrDash(row.keyTrigger) ) ] ) ); // ========================================================= // 2. PERMANENT VERIFIED TRIGGER HISTORY // ========================================================= if ( Array.isArray(row.triggerHistory) && row.triggerHistory.length ) { body.push(`

Permanent Verified Trigger History

${triggerHistoryHtml(row.triggerHistory)}
`); } // ========================================================= // 3. TECHNICAL DETAIL // Only information NOT already shown in main table // ========================================================= const priorMoveStart = firstDefined( row.priorMoveStartDate, row.prior_move_start_date, row.moveStartDate ); const distanceFromBreakout = firstDefined( row.breakoutDistancePct, row.breakout_distance_pct, row.distanceFromBreakoutPct ); const technicalItems = []; if ( priorMoveStart !== null && priorMoveStart !== undefined && priorMoveStart !== "" ) { technicalItems.push( detailItem( "Prior Move Start", escapeHtml(normalizeText(priorMoveStart)) ) ); } if ( distanceFromBreakout !== null && distanceFromBreakout !== undefined && distanceFromBreakout !== "" ) { technicalItems.push( detailItem( "Distance From Breakout %", escapeHtml(formatPct(distanceFromBreakout)) ) ); } if (technicalItems.length) { body.push( detailSection( "Technical Detail", technicalItems ) ); } $("drawerBody").innerHTML = body.join(""); $("drawerBackdrop").classList.add("open"); $("detailDrawer").classList.add("open"); document.body.style.overflow = "hidden"; }
